# UL212 RS-232 Fetch Phase 2 Zigbee Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Zigbee ED uplink to `ul212-rs232-fetch` so valid RS-232 UL212 readings report to the carrier (IDs 16–21), with one compile-time config header for channel and device identity.

**Architecture:** Keep Phase 1 UART driver. Add `zigbee_app_config.h` (channel + ids), a small `zigbee_report` FreeRTOS task that fills `FleetZigbeeConfig` from macros (no NVS/CLI), and enable FleetZigbee in `platformio.ini`. No Bluetooth anywhere. Multi-truck isolation = channel-per-truck.

**Tech Stack:** PlatformIO Arduino-ESP32, FleetProtocol, FleetZigbee, Arduino Zigbee ED, HardwareSerial UART1, USB CDC 115200.

**Spec:** `docs/superpowers/specs/2026-09-04-ul212-rs232-zigbee-design.md`

## Global Constraints

- Board: `seeed_xiao_esp32c6`
- **No Bluetooth / BLE** — do not add BLE libs, GATT, scan, or BLE airtime helpers
- Zigbee identity/RF only via `include/zigbee_app_config.h` (rebuild to change)
- Default: channel **15**, `device_id` **`ul212-rs232-001`**, schema **`1088`**, host_type **`ul212_rs232_fetch`**
- Reading map IDs **16–21** (same keys as BLE host metric map)
- RS-232 default protocol remains **14**; do not change parse driver behaviour except wiring latest reading into Zigbee
- No NVS Preferences for Zigbee config; no USB CLI
- Do **not** `git commit` unless the user explicitly asks
- Keep code short and obvious; prefer few variables

- [ ] **Step 2: Write `zigbee_report.cpp`**

```cpp
#include "zigbee_report.h"

#include "fleet_zigbee_ed.h"
#include "zigbee_app_config.h"

#include <Arduino.h>
#include <string.h>

static portMUX_TYPE s_mux = portMUX_INITIALIZER_UNLOCKED;
static Ul212Reading s_latest{};
static bool s_started;

static const uint32_t ZB_REPORT_PERIOD_MS = 5000;
static const uint32_t ZB_LOOP_MS = 200;

static Ul212Reading copyLatest(void)
{
  Ul212Reading r;
  portENTER_CRITICAL(&s_mux);
  r = s_latest;
  portEXIT_CRITICAL(&s_mux);
  return r;
}

void zigbeeReportUpdate(const Ul212Reading *r)
{
  if (!r || !r->valid) {
    return;
  }
  portENTER_CRITICAL(&s_mux);
  s_latest = *r;
  portEXIT_CRITICAL(&s_mux);
}

static void fillConfig(FleetZigbeeConfig *cfg)
{
  memset(cfg, 0, sizeof(*cfg));
  strncpy(cfg->deviceId, ZB_DEVICE_ID, sizeof(cfg->deviceId) - 1);
  strncpy(cfg->nodeId, ZB_NODE_ID, sizeof(cfg->nodeId) - 1);
  strncpy(cfg->schemaId, ZB_SCHEMA_ID, sizeof(cfg->schemaId) - 1);
  strncpy(cfg->hostType, ZB_HOST_TYPE, sizeof(cfg->hostType) - 1);
  strncpy(cfg->metricMap, ZB_METRIC_MAP, sizeof(cfg->metricMap) - 1);
  cfg->hostTypeId = 1;
  cfg->manifestVersion = 1;
}

static void zigbeeTask(void *param)
{
  (void)param;
  uint32_t last_wait_log_ms = 0;

  for (;;) {
    if (copyLatest().valid) {
      break;
    }
    const uint32_t now = millis();
    if (last_wait_log_ms == 0 || (now - last_wait_log_ms) >= 10000) {
      last_wait_log_ms = now;
      Serial.println("[zb] waiting for first valid RS-232 reading…");
    }
    vTaskDelay(pdMS_TO_TICKS(500));
  }

  FleetZigbeeConfig cfg;
  fillConfig(&cfg);
  Serial.printf("[zb] start device=%s ch=%d\n", cfg.deviceId, FLEET_ZB_CHANNEL);
  fleetZigbeeEdBegin(&cfg);

  uint8_t seq = 0;
  uint32_t last_report_ms = 0;

  for (;;) {
    fleetZigbeeEdLoop();

    const Ul212Reading r = copyLatest();
    const uint32_t now = millis();
    if (fleetZigbeeEdIsJoined() && r.valid &&
        (last_report_ms == 0 || (now - last_report_ms) >= ZB_REPORT_PERIOD_MS)) {
      FleetZigbeeReading readings[6];
      size_t n = 0;
      readings[n++] = {16, FLEET_VAL_FLOAT, {.f32 = r.height_mm}};
      readings[n++] = {17, FLEET_VAL_FLOAT, {.f32 = r.smooth_mm}};
      readings[n++] = {18, FLEET_VAL_FLOAT, {.f32 = r.temp_c}};
      readings[n++] = {19, FLEET_VAL_UINT8, {.u8 = r.signal}};
      readings[n++] = {20, FLEET_VAL_UINT8, {.u8 = (r.signal > 0) ? 1u : 0u}};
      readings[n++] = {21, FLEET_VAL_UINT8, {.u8 = r.tilt_deg}};

      uint8_t status = FLEET_STATUS_SENSOR_CONNECTED;
      if (r.valid) {
        status |= FLEET_STATUS_READING_VALID;
      }

      const bool ok = fleetZigbeeEdSendReport(readings, n, status, ++seq);
      last_report_ms = millis();
      if (ok) {
        Serial.printf("[zb] report seq=%u height=%.1f\n", (unsigned)seq, r.height_mm);
      } else {
        Serial.println("[zb] report TX failed (will retry)");
      }
    }

    vTaskDelay(pdMS_TO_TICKS(ZB_LOOP_MS));
  }
}

void zigbeeReportStart(void)
{
  if (s_started) {
    return;
  }
  s_started = true;
  xTaskCreate(zigbeeTask, "zb_rpt", 8192, nullptr, 5, nullptr);
}
```

Do **not** include any BLE headers or pause helpers.

- [ ] **Step 1: Replace `main.cpp` with**

```cpp
#include <Arduino.h>
#include "board_pins.h"
#include "ul212_rs232.h"
#include "ul212_rs232_config.h"
#include "zigbee_app_config.h"
#include "zigbee_report.h"

void setup() {
  Serial.begin(115200);
  delay(1500);
  Serial.println("=== UL212 RS-232 fetch (XIAO ESP32-C6) ===");
  Serial.printf("protocol=%d address=%d RX=%d TX=%d\n",
                UL212_RS232_PROTOCOL, UL212_RS232_ADDRESS,
                UL212_UART_RX_PIN, UL212_UART_TX_PIN);
  Serial.printf("zigbee device=%s ch=%d (no Bluetooth)\n",
                ZB_DEVICE_ID, FLEET_ZB_CHANNEL);
  ul212Rs232Begin(UL212_UART_RX_PIN, UL212_UART_TX_PIN, 9600);
  zigbeeReportStart();
}

void loop() {
  Ul212Reading r = ul212Rs232Poll(1000);
  if (r.valid) {
    zigbeeReportUpdate(&r);
    Serial.printf("height=%.1f mm  smooth=%.1f mm  temp=%.1f C  tilt=%u deg  signal=%u\n",
                  r.height_mm, r.smooth_mm, r.temp_c, r.tilt_deg, r.signal);
  } else {
    Serial.println("(no frame)");
  }
#if UL212_RS232_PROTOCOL == 51
  delay(200);
#else
  delay(500);
#endif
}
```
